from pathlib import Path

lib.task_for_pid.argtypes=[C.c_uint,C.c_int,C.POINTER(C.c_uint)];lib.task_for_pid.restype=C.c_int
lib.pid_for_task.argtypes=[C.c_uint,C.POINTER(C.c_int)];lib.pid_for_task.restype=C.c_int

python=str(Path(sys.executable).resolve())

try:
 facts['fixture_pid']=child.pid

finally:
