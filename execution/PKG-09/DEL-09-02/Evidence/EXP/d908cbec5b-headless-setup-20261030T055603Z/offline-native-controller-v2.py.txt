def namespace_tree(path):
    """Metadata only, no contents, no following links, only the exact known unique namespace."""
    path = Path(path)
    if not os.path.lexists(path): return None
    result = []
    for current, dirs, files in os.walk(path, followlinks=False):
        for p in [Path(current), *[Path(current) / n for n in sorted(files)]]:
            s = p.lstat()
            require(not stat.S_ISLNK(s.st_mode), 'linked partial namespace: ' + str(p))
            result.append({'path': str(p.relative_to(path)), 'dev': s.st_dev, 'ino': s.st_ino,
                           'mode': stat.S_IMODE(s.st_mode), 'size': s.st_size,
                           'mtime_ns': s.st_mtime_ns, 'ctime_ns': s.st_ctime_ns})
        require(len(result) < 2000, 'unexpected partial namespace expansion')
        for name in dirs: require(not (Path(current) / name).is_symlink(), 'linked partial directory')
    return result

class Run:
    def __init__(self,out,end):
        self.out,self.end=out,end; self.child=None; self.phase=None
        self.lock=threading.RLock(); self.stop=threading.Event(); self.abort=None
        self.peak=0; self.minimum=100; self.supplier_seen=[]; self.ax=None
    def log(self,event,**facts):
        row={'at':now().isoformat(),'phase':self.phase,'event':event,**facts}
        with (self.out/'actions.jsonl').open('a') as f: f.write(json.dumps(row)+'\n')
        print(json.dumps(row),flush=True)
    def alive(self):
        require(self.abort is None,'guard stopped: '+str(self.abort))
        require(now()<self.end,'resource interval ended')
        with self.lock: require(self.child is not None and self.child.poll() is None,'original process ended')
    def monitor(self):
        while not self.stop.wait(0.2):
            rows=[]
            for line in subprocess.check_output(['/bin/ps','-axo','pid=,ppid=,rss=,command='],text=True).splitlines():
                fields=line.strip().split(None,3)
                if len(fields)==4: rows.append((int(fields[0]),int(fields[1]),int(fields[2]),fields[3]))
            own={os.getpid()};
            if self.child: own.add(self.child.pid)
            while True:
                before=len(own); own.update(pid for pid,ppid,rss,cmd in rows if ppid in own)
                if len(own)==before: break
            total=sum(rss for pid,ppid,rss,cmd in rows if pid in own or '/Users/ryan/dev/chirality-t3/' in cmd)
            free=int(subprocess.check_output(['/usr/sbin/sysctl','-n','kern.memorystatus_level'],text=True))
            self.peak=max(self.peak,total); self.minimum=min(self.minimum,free)
            source=[{'pid':pid,'parent':ppid,'command':cmd} for pid,ppid,rss,cmd in rows if pid in own and '/Contents/Resources/codex/' in cmd]
            self.supplier_seen.extend(x for x in source if x not in self.supplier_seen)
            reason=('unexpected supplier in A' if self.phase=='A' and source else 'exclusive.pending' if EXCLUSIVE.exists() else 'shared64GiB' if total>=64*1024*1024 else 'free below22%' if free<22 else 'resource interval ended' if now()>=self.end else None)
            if reason:
                self.abort=reason; self.log('guard_stop',reason=reason); return
    def launch(self,phase,inputs):
        require(self.child is None,'competing original child refused'); self.phase=phase
        require(not EXCLUSIVE.exists() and now()<self.end,'resource admission refused')
        stderr=(self.out/(phase+'-stderr.log')).open('wb'); stdout=(self.out/(phase+'-stdout.log')).open('wb')
        command=['/usr/bin/sandbox-exec','-f',str(ROOT/'witness-inputs/network-denied.sb'),inputs['executable']]
        self.child=subprocess.Popen(command,env=inputs['environment'],cwd=ROOT/'witness-inputs/project',stdout=stdout,stderr=stderr,start_new_session=True)
        require(self.child.pid!=55072,'old process identity refused')
        self.log('original_process_started',pid=self.child.pid,command=command,environment=inputs['environment'])
        self.ax=AX(self.child.pid,self.alive,self.log)
        end=time.monotonic()+5
        while time.monotonic()<end:
            self.alive()
            try: self.ax.main(); break
            except RuntimeError: time.sleep(0.1)
        self.ax.main(); self.ax.snapshot(self.out/(phase+'-initial-AX.json'))
    def settle(self):
        if not self.child: return
        # Cleanup uses only the still-owned unreaped child, never a stored PID/PGID.
        with self.lock: running=self.child.poll() is None
        if running:
            try:
                saved_abort=self.abort; self.abort=None
                # A selector failure still follows the real Quit menu route first.
                self.ax.quit_ui()
                self.abort=saved_abort
            except Exception as e: self.log('quit_UI_unavailable',error=str(e))
            until=min(time.monotonic()+5,time.monotonic()+max(0,(self.end-now()).total_seconds()))
            while time.monotonic()<until:
                with self.lock:
                    if self.child.poll() is not None: break
                time.sleep(0.1)
            with self.lock:
                if self.child.poll() is None:
                    require(not any(x for x in self.supplier_seen),'original App retained: supplier cleanup not proven; do not kill observer')
                    os.killpg(self.child.pid,signal.SIGTERM)
                    self.log('owned_group_signal',signal='TERM',limit='not a normal-quit pass')
                    try: self.child.wait(timeout=2)
                    except subprocess.TimeoutExpired: os.killpg(self.child.pid,signal.SIGKILL); self.child.wait()
        with self.lock: code=self.child.wait()
        self.log('original_process_reaped',pid=self.child.pid,exit_code=code)
        self.child=None; self.ax=None
        return code


def wait_for(run,predicate,seconds,message):
    end=time.monotonic()+seconds
    while time.monotonic()<end:
        run.alive()
        value=predicate()
        if value: return value
        time.sleep(0.1)
    raise RuntimeError(message)
