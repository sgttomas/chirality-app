#[test]
fn v15r1_probe_x2_appends_a_line_its_own_reader_refuses() {
    let s = Scratch::new();
    let one = s.put("sample", false, "One");
    register(&s.persistent_owner(), one.revision());
    {
        let owner = s.persistent_owner();
        let session = owner.review_draft("sample", one.revision()).unwrap();
        let r = receipt(&session);
        let mut attempt = session.begin_hot_registration(r).unwrap();
        STOP_AFTER_STORED.with(|stop| stop.set(true));
        let _ = attempt.advance();
    }
    // A journal whose intended line is schema-valid but breaks RC-9 (sequence 2
    // for a line with sequence 1), as a corrupted or foreign-written file would.
    let dir = s.0.join(".chirality/.workflow-staging/attempts");
    let file = fs::read_dir(&dir).unwrap().filter_map(|e| e.ok()).map(|e| e.path())
        .find(|p| !p.file_name().unwrap().to_string_lossy().starts_with('.')).unwrap();
    let mut journal: Value = serde_json::from_slice(&fs::read(&file).unwrap()).unwrap();
    journal["intended"]["reconfirms"]["sequence"] = json!(2);
    fs::write(&file, serde_json::to_vec(&journal).unwrap()).unwrap();
    assert!(read_ledger(&s.0).is_ok(), "ledger readable before X-2");
    let relaunched = s.persistent_owner();
    eprintln!("PROBE1 reconciliation: {:?}", relaunched.reconciliation());
    let after = read_ledger(&s.0);
    eprintln!("PROBE1 read_ledger after X-2: {:?}", after.as_ref().map(|r| r.len()));
    assert!(after.is_err(), "X-2 appended a line its own reader refuses");
}

#[test]
fn v15r1_probe_second_process_x2_then_continue_gives_two_lines_for_one_act() {
    let s = Scratch::new();
    let one = s.put("sample", false, "One");
    register(&s.persistent_owner(), one.revision());
    let dir = s.0.join(".chirality/.workflow-staging/attempts");
    fs::create_dir_all(&dir).unwrap();
    // Process B: a live DS-8 attempt whose journal is published but whose
    // directory sync fails (production path: create_json -> sync_publication).
    let owner_b = s.persistent_owner();
    let session = owner_b.review_draft("sample", one.revision()).unwrap();
    let r = receipt(&session);
    let record = r.record_id().to_string();
    let mut attempt = session.begin_hot_registration(r).unwrap();
    storage::fail_directory_for_test(Some(dir.clone()));
    let first = attempt.advance();
    storage::fail_directory_for_test(None);
    eprintln!("PROBE2 first advance: {first:?}");
    eprintln!("PROBE2 journals after first advance: {}", fs::read_dir(&dir).unwrap().count());
    // Process C opens the same library: X-2 closes B's live attempt as lost.
    let owner_c = s.persistent_owner();
    eprintln!("PROBE2 process C reconciliation: {:?}", owner_c.reconciliation());
    // B continues its original captured attempt ("Continue original captured registration").
    let second = attempt.advance();
    eprintln!("PROBE2 second advance in B: {second:?}");
    let raw = fs::read_to_string(s.0.join(".chirality/workflow-registry.jsonl")).unwrap();
    let citing = raw.lines().filter(|l| l.contains(&record)).count();
    eprintln!("PROBE2 ledger lines citing the one A15: {citing}");
    let read = read_ledger(&s.0);
    eprintln!("PROBE2 read_ledger: {:?}", read.as_ref().map(|r| r.len()));
    assert!(citing <= 1 && read.is_ok(), "one act, one line; ledger readable");
}

#[test]
fn v15r1_probe_g1r_line_changed_branch_appends_a_line_the_reader_refuses() {
    let s = Scratch::new();
    let one = s.put("sample", false, "One");
    register(&s.persistent_owner(), one.revision());
    let owner = s.persistent_owner();
    let session = owner.review_draft("sample", one.revision()).unwrap();
    let r = receipt(&session);
    // After capture, registered line 1 is edited outside the App (WR G-1R:
    // "line changed or missing -> not completed"): here it no longer reads as
    // registered.
    let path = s.0.join(".chirality/workflow-registry.jsonl");
    let mut line: Value = serde_json::from_str(fs::read_to_string(&path).unwrap().lines().next().unwrap()).unwrap();
    line["outcome"] = json!("not completed");
    line["reason"] = json!("edited outside the App");
    fs::write(&path, format!("{}\n", serde_json::to_string(&line).unwrap())).unwrap();
    assert!(read_ledger(&s.0).is_ok(), "the edited ledger is readable before the attempt");
    let mut attempt = session.begin_hot_registration(r).unwrap();
    let outcome = attempt.advance();
    eprintln!("PROBE5 outcome: {outcome:?}");
    let after = read_ledger(&s.0);
    eprintln!("PROBE5 read_ledger after the attempt: {:?}", after.as_ref().map(|r| r.len()));
    assert!(after.is_ok(), "the App's own not-completed line keeps the ledger readable");
}

#[test]
fn v15r1_probe_p6_append_fails_then_second_process_x2_then_continue() {
    use std::os::unix::fs::PermissionsExt;
    let s = Scratch::new();
    let one = s.put("sample", false, "One");
    register(&s.persistent_owner(), one.revision());
    let path = s.0.join(".chirality/workflow-registry.jsonl");
    let owner_b = s.persistent_owner();
    let session = owner_b.review_draft("sample", one.revision()).unwrap();
    let r = receipt(&session);
    let record = r.record_id().to_string();
    let mut attempt = session.begin_hot_registration(r).unwrap();
    fs::set_permissions(&path, fs::Permissions::from_mode(0o444)).unwrap();
    let first = attempt.advance();
    fs::set_permissions(&path, fs::Permissions::from_mode(0o644)).unwrap();
    eprintln!("PROBE6 first advance (append refused): {first:?}");
    let owner_c = s.persistent_owner();
    eprintln!("PROBE6 process C reconciliation: {:?}", owner_c.reconciliation());
    for n in 0..2 {
        eprintln!("PROBE6 B continue {n}: {:?}", attempt.advance());
    }
    let raw = fs::read_to_string(&path).unwrap();
    let citing = raw.lines().filter(|l| l.contains(&record)).count();
    eprintln!("PROBE6 lines citing the A15: {citing}; read_ledger ok: {}", read_ledger(&s.0).is_ok());
    assert!(citing == 1 && read_ledger(&s.0).is_ok());
}
